async function check(number, fn) {
  const item = checks[number - 1], start = Date.now();
  try {
    const detail = await fn();
    Object.assign(item, { status: 'passed', durationMs: Date.now() - start, ...(detail ? { detail } : {}) });
    console.log(`PASS ${number}: ${item.name}`);
  } catch (error) {
    Object.assign(item, { status: 'failed', durationMs: Date.now() - start, error: error.message });
    throw error;
  }
}

async function context(options = {}) {
  const result = await browser.newContext({ viewport: { width: 1440, height: 900 }, acceptDownloads: true, ...options });
  await result.route('**/*', route => route.request().url().startsWith(base + '/') ? route.continue() : route.abort());
  return result;
}
async function ready(page) {
  currentPage = page;
  await page.waitForFunction(() => document.querySelector('[data-bear-workspace]')?.dataset.ready === 'true');
  await flush(page);
}
async function visit(ctx, route = '/zh/projects/') {
  const page = await ctx.newPage();
  page.setDefaultTimeout(15000);
  page.on('pageerror', error => runtimeErrors.push({ page: page.url(), message: error.message, stack: error.stack }));
  page.on('console', message => {
    if (message.type() === 'error' && /SyntaxError|Unexpected token|Failed to resolve module|does not provide an export|Cannot use import statement/.test(message.text())) runtimeErrors.push({ page: page.url(), message: message.text() });
  });
  await page.goto(base + route, { waitUntil: 'domcontentloaded' });
  await ready(page);
  return page;
}
async function flush(page) {
  // The optional public hook flushes the actual application queue. Each durable
  // assertion below additionally polls localStorage rather than sleeping.
  await page.evaluate(async () => { if (typeof window.BearSpace?.whenSaved === 'function') await window.BearSpace.whenSaved(); });
}
async function disk(page) {
  await flush(page);
  return page.evaluate(key => JSON.parse(localStorage.getItem(key)), storageKey);
}
async function waitNote(page, note) {
  await flush(page);
  await page.waitForFunction(({ key, note }) => {
    try { return JSON.parse(localStorage.getItem(key))?.note === note; } catch { return false; }
  }, { key: storageKey, note });
}
function quotedText(markdown) {
  return markdown.split('\n').filter(line => line.startsWith('> ')).map(line => line.slice(2).replace(/\\([\\`*_[\]{}()!~|])/g,'$1').replace(/&lt;/g,'<').replace(/&gt;/g,'>').replace(/&amp;/g,'&')).join('\n');
}
async function snapshot(page) { return page.evaluate(() => window.BearSpace.snapshot()); }
async function open(page, app, payload) {
  currentPage = page;
  await page.evaluate(({ app, payload }) => window.BearSpace.execute('open', { app, ...(payload === undefined ? {} : { payload }) }), { app, payload });
  await page.locator(`[data-space-window="${app}"]`).waitFor({ state: 'visible' });
  await flush(page);
}
async function maximize(page, app) {
  if (await page.evaluate(app => !window.BearSpace.snapshot().windows.find(window => window.app === app)?.maximized, app)) await page.locator(`[data-space-window="${app}"] [data-window-action="maximize"]`).click();
  await flush(page);
}
async function catalog(page) { return page.locator('[data-space-catalog]').evaluate(el => JSON.parse(el.textContent)); }
async function realArticle(page) {
  const data = await catalog(page);
  const preferred = data.articles.find(article => article.url.includes('japan-travel-2025'));
  const candidates = [...(preferred ? [preferred] : []), ...data.articles.slice(0, 8).filter(article => article !== preferred)];
  for (const article of candidates) {
    await open(page, 'reader', article.url);
    await maximize(page, 'reader');
    try {
      await page.locator('[data-reader-body] .post-content').waitFor();
      const usable = await page.locator('[data-reader-body] .post-content').evaluate(el => [...el.querySelectorAll('p')].some(p => p.textContent.trim().length >= 40));
      if (usable) return article;
    } catch { /* Try another real published article; never replace its response. */ }
  }
  throw new Error(`No readable published article found in ${data.language} catalog`);
}
async function selectActualText(page) {
  const text = await page.locator('[data-reader-body] .post-content').evaluate(body => {
    const walker = document.createTreeWalker(body, NodeFilter.SHOW_TEXT);
    let textNode;
    while ((textNode = walker.nextNode())) {
      if (textNode.parentElement?.closest('p') && !textNode.parentElement.closest('pre,code,a') && textNode.textContent.trim().length >= 40) break;
    }
    if (!textNode) throw new Error('No actual article paragraph available for selection');
    const start = textNode.textContent.search(/\S/), range = document.createRange();
    range.setStart(textNode, start); range.setEnd(textNode, Math.min(textNode.textContent.length, start + 96));
    const selection = document.getSelection(); selection.removeAllRanges(); selection.addRange(range);
    return range.toString().replace(/\r\n?/g, '\n').trim();
  });
  await page.waitForFunction(() => document.querySelector('[data-excerpt-save]')?.disabled === false);
  return text;
}
async function saveSelected(page, article, keyboard = false) {
  const text = await selectActualText(page), button = page.locator('[data-excerpt-save]');
  if (keyboard) { await button.focus(); await button.press('Enter'); } else await button.click();
  await flush(page);
  await page.waitForFunction(({ key, text, url }) => {
    try { return JSON.parse(localStorage.getItem(key))?.excerpts?.some(item => item.text === text && item.url === url); } catch { return false; }
  }, { key: storageKey, text, url: article.url });
  const item = (await disk(page)).excerpts.find(item => item.text === text && item.url === article.url);
  assert.equal(item.title, article.title, 'the source title must come from the actual catalog');
  return item;
}
async function capture(page, filename) { await page.screenshot({ path: path.join(outputDir, filename), fullPage: false }); }
async function downloadJSON(page, action, filename) {
  const pending = page.waitForEvent('download'); await action();
  const download = await pending, destination = path.join(outputDir, filename);
  await download.saveAs(destination);
  return JSON.parse(await fs.readFile(destination, 'utf8'));
}

  await page.locator('[data-transfer-preview]').waitFor({ state: 'visible' });
  assert.equal(await page.locator('[data-transfer-confirm]').isEnabled(), true);
}

    }
  });
  await check(15, async () => assert.deepEqual(runtimeErrors, [], 'browser JavaScript errors must not be hidden by functional checks'));
} catch (error) {
  fatal = { message: error.message, stack: error.stack };
  process.exitCode = 1;
  console.error(error.stack || error.message);
  if (currentPage && !currentPage.isClosed()) await capture(currentPage, 'failure.png').catch(() => {});
} finally {
  const report = { startedAt, completedAt: new Date().toISOString(), base, publicDir, outputDir, checks, runtimeErrors, fatal, passed: checks.filter(item => item.status === 'passed').length, total: checks.length };
  try { await fs.writeFile(path.join(outputDir, 'checks.json'), JSON.stringify(report, null, 2)); }
  finally {
    // Only the browser and frozen-output server created by this script are closed.
    try { if (browser) await browser.close(); }
    finally { if (server?.listening) await new Promise(resolve => server.close(resolve)); }
  }
}
