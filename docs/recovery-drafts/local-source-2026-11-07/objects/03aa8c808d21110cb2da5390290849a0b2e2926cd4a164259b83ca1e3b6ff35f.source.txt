test('native resume can take the stream after its coordinator pause was confirmed', async () => {
  const environment = setupEnvironment();
  const hostA = new FakeElement('div', environment.doc), hostB = new FakeElement('div', environment.doc);
  const statuses = [];
  let a, b;
  try {
    environment.win.YT = { Player: FakePlayer };
    a = mountVideoPlayer({host:hostA,reference:WATCH,language:'en',onStatus:s=>statuses.push(s)});
    b = mountVideoPlayer({host:hostB,reference:WATCH,language:'en'});
    hostA.querySelector('[data-video-play]').click(); await tick(); await tick();
    const playerA = FakePlayer.instances.at(-1); playerA.emit('onReady'); playerA.emit('onStateChange',{data:1});
    hostB.querySelector('[data-video-play]').click(); await tick(); await tick();
    const playerB = FakePlayer.instances.at(-1); playerB.emit('onReady'); playerB.emit('onStateChange',{data:1});
    playerA.emit('onStateChange',{data:2}); // pause acknowledged before a new native gesture
    const pauses = playerB.calls.filter(call=>call==='pause').length;
    playerA.emit('onStateChange',{data:1});
    assert.equal(statuses.at(-1).status,'playing');
    assert.equal(playerB.calls.filter(call=>call==='pause').length,pauses+1);
  } finally { a?.dispose(); b?.dispose(); environment.restore(); }
});

class FakeText {
  constructor(value) { this.nodeType = 3; this.textContent = String(value); this.parentNode = null; }
}
function matchesSimple(node, part) {
  const parsed = /^([A-Za-z0-9-]*)((?:\[[^\]]+\])*)$/.exec(part);
  if (!parsed) return false;
  const tag = parsed[1].toLowerCase();
  if (tag && node.tagName !== tag.toUpperCase()) return false;
  for (const attribute of parsed[2].matchAll(/\[([A-Za-z0-9_-]+)(?:=("?)([^\]"]*)\2)?\]/g)) {
    const name = attribute[1].toLowerCase();
    if (!node.hasAttribute(name)) return false;
    if (attribute[3] !== undefined && node.getAttribute(name) !== attribute[3]) return false;
  }
  return true;
}
class FakeElement extends EventTarget {
  constructor(tagName, ownerDocument) {
    super();
    this.nodeType = 1;
    this.tagName = String(tagName).toUpperCase();
    this.ownerDocument = ownerDocument;
    this.parentNode = null;
    this.childNodes = [];
    this.attributes = new Map();
    this.hidden = false;
    this.className = '';
    this.scrollRequests = [];
  }
  set textContent(value) { this.childNodes = []; if (value !== '') this.appendChild(new FakeText(value)); }
  get textContent() { return this.childNodes.map(node => node.textContent).join(''); }
  get children() { return this.childNodes.filter(node => node.nodeType === 1); }
  appendChild(node) {
    if (node.parentNode) node.parentNode.removeChild(node);
    node.parentNode = this; this.childNodes.push(node); return node;
  }
  insertBefore(node, reference) {
    if (node.parentNode) node.parentNode.removeChild(node);
    const index = this.childNodes.indexOf(reference);
    node.parentNode = this;
    if (index === -1) this.childNodes.push(node); else this.childNodes.splice(index, 0, node);
    return node;
  }
  replaceChildren(...nodes) { this.childNodes = []; for (const node of nodes) this.appendChild(node); }
  removeChild(node) { const index = this.childNodes.indexOf(node); if (index >= 0) this.childNodes.splice(index, 1); node.parentNode = null; return node; }
  remove() { if (this.parentNode) this.parentNode.removeChild(this); }
  setAttribute(name, value) { this.attributes.set(String(name).toLowerCase(), String(value)); }
  getAttribute(name) { const value = this.attributes.get(String(name).toLowerCase()); return value === undefined ? null : value; }
  hasAttribute(name) { return this.attributes.has(String(name).toLowerCase()); }
  removeAttribute(name) { this.attributes.delete(String(name).toLowerCase()); }
  matches(selector) { return String(selector).split(',').some(part => matchesSimple(this, part.trim())); }
  closest(selector) { let node = this; while (node && node.nodeType === 1) { if (node.matches(selector)) return node; node = node.parentNode; } return null; }
  querySelectorAll(selector) {
    const found = [];
    const walk = node => { for (const child of node.childNodes) { if (child.nodeType === 1) { if (child.matches(selector)) found.push(child); walk(child); } } };
    walk(this);
    return found;
  }
  querySelector(selector) { return this.querySelectorAll(selector)[0] || null; }
  click() { this.dispatchEvent(new Event('click')); }
  scrollIntoView(options) { this.scrollRequests.push(options); }
}
class FakeDocument extends EventTarget {
  constructor() {
    super();
    this.hidden = false;
    this.documentElement = new FakeElement('html', this);
    this.head = new FakeElement('head', this);
    this.body = new FakeElement('body', this);
    this.documentElement.appendChild(this.head);
    this.documentElement.appendChild(this.body);
  }
  createElement(tagName) { return new FakeElement(tagName, this); }
}

test('parseYouTubeURL accepts the exact HTTPS YouTube forms and canonicalises them', () => {
  for (const value of [
    `https://www.youtube.com/watch?v=${ID}`,
    `https://youtube.com/watch?v=${ID}`,
    `https://m.youtube.com/watch?v=${ID}`,
    `https://www.youtube.com/watch/?v=${ID}`,
    `https://www.youtube.com/embed/${ID}`,
    `https://www.youtube.com/shorts/${ID}`,
    `https://youtu.be/${ID}`,
    `https://youtu.be/${ID}/`,
    `HTTPS://WWW.YOUTUBE.COM/watch?v=${ID}`
  ]) {
    const parsed = parseYouTubeURL(value);
    assert.ok(parsed, value);
    assert.equal(parsed.id, ID, value);
    assert.equal(parsed.start, 0, value);
    assert.equal(parsed.watchURL, WATCH, value);
    assert.deepEqual(Object.keys(parsed).sort(), ['id', 'start', 'watchURL'], value);
  }
});

test('start times parse from t/start in seconds or hms within 0..21600', () => {
  for (const [raw, expected] of [['0', 0], ['90', 90], ['90s', 90], ['1m30s', 90], ['1h2m3s', 3723], ['1h', 3600], ['21600', 21600], ['6h', 21600], ['1:30', 90], ['1:02:03', 3723]]) {
    for (const template of [`https://www.youtube.com/watch?v=${ID}&t=${raw}`, `https://www.youtube.com/watch?v=${ID}&start=${raw}`, `https://www.youtube.com/embed/${ID}?start=${raw}`, `https://youtu.be/${ID}?t=${raw}`]) {
      const parsed = parseYouTubeURL(template);
      assert.ok(parsed, template);
      assert.equal(parsed.start, expected, template);
      assert.equal(parsed.watchURL, expected > 0 ? `${WATCH}&t=${expected}s` : WATCH, template);
    }
  }
  for (const raw of ['21601', '6h1s', '-5', '1.5', '90x', 'abc', '', '1m30', '1:2:3:4', '1000000', '99:99']) {
    assert.equal(parseYouTubeURL(`https://www.youtube.com/watch?v=${ID}&t=${encodeURIComponent(raw)}`), null, `t=${raw}`);
    assert.equal(parseYouTubeURL(`https://www.youtube.com/watch?v=${ID}&start=${encodeURIComponent(raw)}`), null, `start=${raw}`);
  }
  assert.equal(MAX_VIDEO_START_SECONDS, 21600);
});

test('canonical watch URLs round-trip and reference helpers stay inert', () => {
  const parsed = parseYouTubeURL(`https://www.youtube.com/embed/${ID}?start=3723`);
  assert.equal(parsed.watchURL, `${WATCH}&t=3723s`);
  const roundTrip = parseYouTubeURL(parsed.watchURL);
  assert.deepEqual({ id: roundTrip.id, start: roundTrip.start }, { id: parsed.id, start: parsed.start });
  assert.equal(inertTitle('  有\u0000控制\n字符  ', 'fallback'), '有 控制 字符');
  assert.equal(inertTitle('', 'fallback'), 'fallback');
  assert.equal(inertTitle('x'.repeat(300)).length, 120);
  assert.deepEqual(clampVideoSize(10, 999999), { width: 200, height: 2160 });
  assert.deepEqual(clampVideoSize(640, 360), { width: 640, height: 360 });
  assert.deepEqual(clampVideoSize('nope', NaN), { width: 640, height: 360 });
  const local = '/images/posts/ai-english-learning/youtube-evals-english-transcript.png';
  const normalized = normalizeVideoReference({ id: ID, title: ' 标题 ', poster: local }, [{ id: ID, title: 'References title' }]);
  assert.deepEqual({ id: normalized.id, start: normalized.start, watchURL: normalized.watchURL, title: normalized.title, poster: normalized.poster },
    { id: ID, start: 0, watchURL: WATCH, title: '标题', poster: local });
  const fromReferences = normalizeVideoReference(WATCH, [{ id: ID, title: 'Authoritative title', poster: local }]);
  assert.equal(fromReferences.title, 'Authoritative title');
  assert.equal(normalizeVideoReference(WATCH, [], 'anchor label').title, 'anchor label');
  for (const poster of ['https://i.ytimg.com/vi/x/hq.jpg', '//cdn.test/x.jpg', '/img\\x.jpg', 'relative.jpg']) {
    assert.equal(normalizeVideoReference({ id: ID, poster }, []).poster, '', poster);
  }
  const known = KNOWN_VIDEO_REFERENCES.find(item => item.id === ID);
  assert.ok(known, 'the real known video is registered');
  assert.equal(known.article, '/zh/growth/posts/ai-english-learning/');
  assert.equal(normalizeVideoReference(WATCH, []).poster, known.poster);
});

test('the iframe load alone is never ready or playing; ready comes from the API event', async () => {
  const fixture = mountWindowPlayer();
  try {
    fixture.win.YT = { Player: FakePlayer };
    fixture.play();
    const player = await fixture.ready();
    assert.ok(player);
    assert.equal(fixture.statuses.some(status => status.status === 'ready' || status.status === 'playing'), false);
    assert.equal(player.calls.includes('play'), false);
    player.emit('onReady');
    assert.equal(fixture.statuses.at(-1).status, 'ready');
    assert.equal(player.calls.filter(call => call === 'play').length, 1);
    player.emit('onStateChange', { data: 3 });
    assert.equal(fixture.statuses.at(-1).status, 'buffering');
    player.emit('onStateChange', { data: 1 });
    assert.equal(fixture.statuses.at(-1).status, 'playing');
    player.emit('onStateChange', { data: 2 });
    assert.equal(fixture.statuses.at(-1).status, 'paused');
    player.emit('onStateChange', { data: 0 });
    assert.equal(fixture.statuses.at(-1).status, 'ended');
  } finally { fixture.cleanup(); }
});

    widgetB = mountVideoPlayer({ host: hostB, reference: WATCH, language: 'zh', onStatus() {} });

    hostA.querySelector('[data-video-play]').click();
    await tick();
    await tick();
    const playerA = FakePlayer.instances.at(-1);
    playerA.emit('onReady');
    playerA.emit('onStateChange', { data: 1 });
    assert.deepEqual(playerA.calls, ['play']);
    // B claims the stream with an explicit play; A is paused.
    hostB.querySelector('[data-video-play]').click();
    assert.deepEqual(playerA.calls, ['play', 'pause']);
    await tick();
    await tick();
    const playerB = FakePlayer.instances.at(-1);

    playerA.emit('onStateChange', { data: 2 }); // coordinator pause acknowledged
    const pausesBefore = playerA.calls.filter(call => call === 'pause').length;
    playerA.emit('onStateChange', { data: 1 });
    assert.equal(playerA.calls.filter(call => call === 'pause').length, pausesBefore); // a claim never pauses itself
    assert.equal(playerB.calls.filter(call => call === 'pause').length, 2);

    widgetA.dispose();
    widgetB.dispose();

test('another chapter gesture cancels the older widget pending seek and claims its stream', async () => {
  const environment = setupEnvironment();
  const hostA = new FakeElement('div', environment.doc), hostB = new FakeElement('div', environment.doc);
  let a, b;
  try {
    environment.win.YT = { Player: FakePlayer };
    a = mountVideoPlayer({ host: hostA, reference: chapterReference() });
    b = mountVideoPlayer({ host: hostB, reference: chapterReference() });
    chapterButton(hostA, 1011).click();
    await tick(); await tick();
    const playerA = FakePlayer.instances.at(-1);
    chapterButton(hostA, 1899).click();
    chapterButton(hostB, 1434).click();
    await tick(); await tick();
    const playerB = FakePlayer.instances.at(-1);
    playerA.emit('onReady');
    assert.deepEqual(playerA.seeks, []);
    assert.equal(playerA.calls.includes('play'), false);
    playerB.emit('onReady');
    playerB.emit('onStateChange', { data: 1 });
    chapterButton(hostA, 1434).click();
    assert.deepEqual(playerA.seeks, [{ seconds: 1434, allowSeekAhead: true }]);
    assert.equal(playerA.calls.filter(call => call === 'play').length, 1);
    assert.equal(playerB.calls.filter(call => call === 'pause').length, 1);
  } finally { a?.dispose(); b?.dispose(); environment.restore(); }
});

test('a failed chapter seek does not play the wrong moment and offers a retry at the selected time', async () => {
  const fixture = mountWindowPlayer({ reference: chapterReference() });
  try {
    fixture.win.YT = { Player: FakePlayer };
    fixture.play();
    const player = await fixture.ready();
    player.emit('onReady');
    player.emit('onStateChange', { data: 2 });
    player.seekTo = () => { throw new Error('seek rejected'); };
    chapterButton(fixture.host, 1434).click();
    assert.equal(player.calls.filter(call => call === 'play').length, 1);
    assert.equal(fixture.statuses.at(-1).status, 'error');
    assert.match(fixture.statuses.at(-1).message, /无法跳转到所选片段/);
    assert.equal(fixture.host.querySelector('[data-video-notice-source]').getAttribute('href'), `${WATCH}&t=1434s`);
    fixture.host.querySelector('[data-video-retry]').click();
    const replacement = await fixture.ready();
    assert.equal(new URL(replacement.element.getAttribute('src')).searchParams.get('start'), '1434');
    replacement.emit('onReady');
    assert.deepEqual(replacement.calls, ['play']);
  } finally { fixture.cleanup(); }
});
