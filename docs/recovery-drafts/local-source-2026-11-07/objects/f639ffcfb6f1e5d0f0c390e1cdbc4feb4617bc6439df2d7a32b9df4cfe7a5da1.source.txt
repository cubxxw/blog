const moduleURL = new URL('../assets/js/space/media.js', import.meta.url);
const source = (await readFile(moduleURL, 'utf8')).replace(/from '\.\/(state\.mjs|media-state\.mjs)'/g, (_match, file) => `from '${new URL(file, moduleURL).href}'`);

// A minimal DOM boundary exercises the real mounted load/dispose handlers without a browser service.
class Button extends EventTarget {
  constructor(dataset = {}) { super(); this.dataset = dataset; }
  click() { this.dispatchEvent(new Event('click')); }
}
class Host extends EventTarget {
  constructor() { super(); this.isConnected = true; this.markup = ''; }
  set innerHTML(value) {
    this.markup = value; this.load = new Button();
    this.more = [...value.matchAll(/data-media-more="([^"]+)"/g)].map(match => new Button({ mediaMore: match[1] }));
    this.routes = [...value.matchAll(/<button\b([^>]*data-media-open="[^"]+"[^>]*)>/g)].map(match => {
      const attributes = Object.fromEntries([...match[1].matchAll(/([a-z-]+)="([^"]*)"/g)].map(attribute => [attribute[1], attribute[2]]));
      const button = new Button({ mediaOpen: attributes['data-media-open'], mediaPayload: attributes['data-media-payload'] });
      button.className = attributes.class || '';
      return button;
    });
    this.placeChoices = [...value.matchAll(/data-place-select="([^"]+)"/g)].map(match => new Button({ placeSelect: match[1] }));
    this.zoomChoices = [...value.matchAll(/data-map-zoom="([^"]+)"/g)].map(match => new Button({ mapZoom: match[1] }));
    this.back = /data-photo-back/.test(value) ? new Button() : null;
    this.viewport = new EventTarget();
    this.viewport.querySelector = () => null;
    this.viewport.dataset = {};
  }
  get innerHTML() { return this.markup; }
  querySelectorAll(selector) { return selector === '[data-media-load]' ? [this.load] : selector === '[data-media-more]' ? this.more : selector === '[data-media-open]' ? this.routes : selector === '[data-place-select]' ? this.placeChoices : selector === '[data-map-zoom]' ? this.zoomChoices : []; }
  querySelector(selector) { return selector === '[data-map-viewport]' ? this.viewport : selector === '[data-photo-back]' ? this.back : null; }
  closest() { return null; }
}
const ID = suffix => `${suffix.repeat(8)}-${suffix.repeat(4)}-4${suffix.repeat(3)}-8${suffix.repeat(3)}-${suffix.repeat(12)}`;
const remote = (suffix, album = 'journey', title = suffix) => ({ id: `immich:${album}:${ID(suffix)}`, album, title, place: 'Public album', region: null, source: 'immich' });
const album = (slug = 'journey', nextCursor = null) => ({ slug, title: slug, nextCursor });
const response = (photos, albums = [album()], extra = {}) => new Response(JSON.stringify({ connected: true, photos, albums, reason: 'connected', ...extra }), { headers: { 'Content-Type': 'application/json' }, status: extra.status || 200 });
const staticPhoto = { id: 'kumano', title: 'Kumano', place: 'Japan', image: '/public.webp', thumb: '/public.webp', region: 'japan' };
function setup(fetchImpl, options = {}) {
  const originalFetch = globalThis.fetch, originalLocation = globalThis.location;
  globalThis.fetch = fetchImpl; globalThis.location = { origin: 'https://blog.test' };
  const catalog = { language: 'en', photos: [staticPhoto], places: [], articles: [] }, notifications = [];
  const media = mountMedia({ catalog, execute() {}, notify: text => notifications.push(text), ...options });
  const host = new Host(); media.renderPhotos(host, '', 'one');
  return { media, catalog, notifications, host, cleanup() { media.dispose('one'); media.dispose('two'); globalThis.fetch = originalFetch; globalThis.location = originalLocation; } };
}
async function until(predicate) {
  const deadline = Date.now() + 1000;
  while (!predicate()) { if (Date.now() > deadline) assert.fail('load did not settle'); await new Promise(resolve => setImmediate(resolve)); }
}
function deferred() { let resolve; const promise = new Promise(r => { resolve = r; }); return { promise, resolve }; }

test('full refresh replaces remote projection and preserves the static object and array', async () => {
  let round = 0;
  const fixture = setup(() => response(round++ ? [remote('2', 'journey', 'updated'), remote('3')] : [remote('1'), remote('2')]));
  try {
    const photosArray = fixture.catalog.photos;
    fixture.host.load.click(); await until(() => fixture.media.status().phase === 'ready');
    assert.deepEqual(fixture.catalog.photos.map(p => p.id), ['kumano', remote('1').id, remote('2').id]);
    fixture.host.load.click(); await until(() => fixture.notifications.length === 2);
    assert.equal(fixture.catalog.photos, photosArray); assert.equal(fixture.catalog.photos[0], staticPhoto);
    assert.deepEqual(fixture.catalog.photos.map(p => p.id), ['kumano', remote('2').id, remote('3').id]);
    assert.equal(fixture.catalog.photos[1].title, 'updated');
  } finally { fixture.cleanup(); }
});
test('pagination appends the requested album, updates duplicate IDs and retains other albums', async () => {
  let round = 0, pagingURL;
  const fixture = setup(url => { pagingURL = url; return round++ ? response([remote('1', 'journey', 'updated'), remote('3'), remote('4', 'other')], [album('journey')]) : response([remote('1'), remote('2', 'other')], [album('journey', 'a'.repeat(50)), album('other')]); });
  try {
    fixture.host.load.click(); await until(() => fixture.media.status().phase === 'ready');
    fixture.host.more[0].click(); await until(() => fixture.notifications.length === 2);
    assert.ok(pagingURL.includes('album=journey&cursor='));
    assert.deepEqual(new Set(fixture.catalog.photos.map(p => p.id)), new Set(['kumano', remote('1').id, remote('2', 'other').id, remote('3').id]));
    assert.equal(fixture.catalog.photos.find(p => p.id === remote('1').id).title, 'updated');
  } finally { fixture.cleanup(); }
});

test('disposing the last media window aborts transport and ignores late results', async () => {
  const pending = deferred(); let signal;
  const fixture = setup((_url, options) => { signal = options.signal; return pending.promise; });
  try {
    fixture.host.load.click(); fixture.media.dispose('one');
    assert.equal(signal.aborted, true); assert.equal(fixture.media.status().reason, 'cancelled');
    pending.resolve(response([remote('1')])); await new Promise(resolve => setImmediate(resolve));
    assert.deepEqual(fixture.catalog.photos, [staticPhoto]); assert.deepEqual(fixture.notifications, []); assert.equal(fixture.media.status().reason, 'cancelled');
  } finally { fixture.cleanup(); }
});
test('late cancelled request cannot overwrite a newly reopened media load', async () => {

    assert.deepEqual(fixture.catalog.photos.map(p => p.id), ['kumano', remote('2').id]); assert.equal(fixture.notifications.length, 1); assert.equal(fixture.media.status().phase, 'ready');
  } finally { fixture.cleanup(); }
});

function navigationFixture(fetchImpl) {
  const savedLocation = globalThis.location, savedFetch = globalThis.fetch;
  globalThis.location = { origin: 'https://blog.test' };
  let fetches = 0;
  globalThis.fetch = (...args) => { fetches++; if (fetchImpl) return fetchImpl(...args); throw new Error('navigation must not load albums'); };
  const catalog = { language: 'en', articles: [], photos: [staticPhoto, { ...staticPhoto, id: 'fuji', title: 'Mount Fuji' }], places: [{ id: 'japan', name: 'Japan', lon: 138.73, lat: 35.36 }, { id: 'southeast-asia', name: 'Southeast Asia', lon: 102.45, lat: 18.92 }] };
  let photoHost = new Host(), mapHost = new Host();
  const calls = [];
  const media = mountMedia({ catalog, execute(command, args) {
    calls.push({ command, ...args });
    if (args.app === 'photos') media.renderPhotos(photoHost, args.payload, 'photos');
    if (args.app === 'map') media.renderMap(mapHost, args.payload, 'map');
  }, notify() {} });
  return {
    catalog, media, calls,
    get photoHost() { return photoHost; }, get mapHost() { return mapHost; },
    closePhotos() { media.dispose('photos'); photoHost.isConnected = false; photoHost = new Host(); },
    closeMap() { media.dispose('map'); mapHost.isConnected = false; mapHost = new Host(); },
    get fetches() { return fetches; },
    cleanup() { media.dispose('photos'); media.dispose('map'); globalThis.location = savedLocation; globalThis.fetch = savedFetch; }
  };
}
const returnRoute = host => host.routes.find(button => button.className.includes('sp-media-return'));

test('both photo-detail map links carry the exact source and filter, and return works after closing Photos', () => {
  for (const region of ['all', 'japan']) {
    const fixture = navigationFixture();
    try {
      fixture.media.renderPhotos(fixture.photoHost, `region:${region}`, 'photos');
      fixture.media.renderPhotos(fixture.photoHost, 'fuji', 'photos');
      const links = fixture.photoHost.routes.filter(button => button.dataset.mediaOpen === 'map');
      assert.equal(links.length, 2);
      assert.equal(links[0].dataset.mediaPayload, `photo-context:fuji:${region}`);
      assert.equal(links[1].dataset.mediaPayload, links[0].dataset.mediaPayload);
      links[0].click();
      assert.ok(returnRoute(fixture.mapHost));
      assert.match(fixture.mapHost.innerHTML, /Back to this photo: Mount Fuji/);
      assert.ok(fixture.mapHost.routes.some(button => button.dataset.mediaPayload === 'region:japan'));
      fixture.media.renderPhotos(fixture.photoHost, 'region:southeast-asia', 'photos');
      fixture.closePhotos();
      returnRoute(fixture.mapHost).click();
      assert.match(fixture.photoHost.innerHTML, /sp-photo-detail/);
      assert.match(fixture.photoHost.innerHTML, /Mount Fuji/);
      fixture.photoHost.back.click();
      assert.equal(fixture.calls.at(-1).payload, `region:${region}`);
      assert.equal(fixture.fetches, 0);
    } finally { fixture.cleanup(); }
  }
});

test('map context revalidates on reopen while plain entries and explicit region choices clear the source', () => {
  const fixture = navigationFixture();
  try {
    const context = 'photo-context:fuji:all';
    fixture.media.renderMap(fixture.mapHost, context, 'map');
    assert.ok(returnRoute(fixture.mapHost));
    fixture.closeMap();
    fixture.media.renderMap(fixture.mapHost, context, 'map');
    assert.ok(returnRoute(fixture.mapHost));
    fixture.mapHost.placeChoices.find(button => button.dataset.placeSelect === 'japan').click();
    assert.equal(returnRoute(fixture.mapHost), undefined, 'even selecting the same region explicitly clears the source');
    fixture.media.renderMap(fixture.mapHost, context, 'map');
    fixture.mapHost.placeChoices.find(button => button.dataset.placeSelect === 'southeast-asia').click();
    assert.equal(returnRoute(fixture.mapHost), undefined);
    fixture.media.renderMap(fixture.mapHost, 'japan', 'map');
    assert.equal(returnRoute(fixture.mapHost), undefined);
    assert.doesNotMatch(fixture.mapHost.innerHTML, /data-media-origin-unavailable/);
    assert.equal(fixture.fetches, 0);
  } finally { fixture.cleanup(); }
});

test('an unavailable public source is revalidated on load and a stale return cannot display another photograph', () => {
  const fixture = navigationFixture();
  try {
    const photo = { ...remote('1'), region: 'japan', image: '/public.webp', thumb: '/public.webp' };
    const context = `photo-context:${encodeURIComponent(photo.id)}:all`;
    fixture.media.renderMap(fixture.mapHost, context, 'map');
    assert.equal(returnRoute(fixture.mapHost), undefined);
    fixture.catalog.photos.push(photo);
    fixture.media.renderMap(fixture.mapHost, context, 'map');
    const back = returnRoute(fixture.mapHost);
    assert.ok(back);
    fixture.catalog.photos.splice(fixture.catalog.photos.indexOf(photo), 1);
    back.click();
    assert.doesNotMatch(fixture.photoHost.innerHTML, /sp-photo-detail/);
    assert.match(fixture.photoHost.innerHTML, /data-media-origin-unavailable/);
    fixture.media.renderMap(fixture.mapHost, context, 'map');
    assert.equal(returnRoute(fixture.mapHost), undefined);
    assert.equal(fixture.fetches, 0);
  } finally { fixture.cleanup(); }
});

test('loading a restored public source corrects the default map region during redraw without rerender navigation', async () => {
  const photo = { ...remote('1'), region: 'southeast-asia', image: '/public.webp', thumb: '/public.webp' };
  const context = `photo-context:${encodeURIComponent(photo.id)}:all`;
  const fixture = navigationFixture(() => response([photo]));
  try {
    fixture.media.renderMap(fixture.mapHost, context, 'map');
    assert.equal(/data-place-select="([^"]+)" aria-pressed="true"/.exec(fixture.mapHost.innerHTML)?.[1], 'japan');
    assert.equal(returnRoute(fixture.mapHost), undefined);
    fixture.media.renderPhotos(fixture.photoHost, '', 'photos');
    fixture.photoHost.load.click();
    await until(() => fixture.media.status().phase === 'ready');
    assert.equal(/data-place-select="([^"]+)" aria-pressed="true"/.exec(fixture.mapHost.innerHTML)?.[1], 'southeast-asia');
    assert.ok(returnRoute(fixture.mapHost));
    assert.equal(returnRoute(fixture.mapHost).dataset.mediaPayload, context);
    assert.doesNotMatch(fixture.mapHost.innerHTML, /data-media-origin-unavailable/);
    assert.equal(fixture.calls.length, 0, 'album redraw must not need another open or renderMap call');
    returnRoute(fixture.mapHost).click();
    assert.match(fixture.photoHost.innerHTML, /sp-photo-detail/);
    fixture.photoHost.back.click();
    assert.equal(fixture.calls.at(-1).payload, 'region:all');
    assert.equal(fixture.fetches, 1);
  } finally { fixture.cleanup(); }
});
